function tgErr(msg){const m=String(msg||'');
  if(/TIMEOUT/.test(m))return 'Telegram не отвечает дольше 10 секунд (возможна блокировка в вашей сети, попробуйте VPN или другой интернет).';
  if(/Failed to fetch|NetworkError|Load failed/i.test(m))return 'нет связи с Telegram (проверьте интернет и блокировки; в превью Claude запросы наружу запрещены).';
  if(/Unauthorized|\(401\)/i.test(m))return 'неверный токен бота.';
  if(/chat not found|\(400\)/i.test(m))return 'неверный chat id.';
  if(/Forbidden|\(403\)/i.test(m))return 'бот не может написать администратору: откройте бота в Telegram и нажмите /start.';
  return m||'неизвестная ошибка.';}
async function sendTG(text){
  let tm;
  const timeout=new Promise((_,rej)=>{tm=setTimeout(()=>rej(new Error('TIMEOUT')),10000)});/* таймаут 10 с без AbortController: в iOS WebView сигнал не клонируется */
  try{
    const r=await Promise.race([fetch('https://api.telegram.org/bot'+TELEGRAM_BOT_TOKEN+'/sendMessage',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({chat_id:TELEGRAM_ADMIN_CHAT_ID,text:text})}),timeout]);
    const j=await r.json().catch(()=>({}));if(!j.ok)throw new Error((j.description||'Ошибка Telegram')+' ('+r.status+')');return 'ok';
  }finally{clearTimeout(tm)}
}
